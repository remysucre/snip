def fib(n):
    a, b = 0, 1
    for _ in range(n):
        a, b = b, a + b
    return a

for i in range(8):
    print(i, fib(i))

counts = {}
for word in 'the quick brown fox the lazy dog the end'.split():
    counts[word] = counts.get(word, 0) + 1

for word, n in sorted(counts.items(), key=lambda kv: -kv[1]):
    print(f'{word:6} {n}')
